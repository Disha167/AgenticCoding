package com.practice.watchlist.bulk;

import java.sql.Connection;
import java.sql.Statement;
import java.util.List;

public class BulkWatchImportController {

    private final BulkWatchImportService service;

    public BulkWatchImportController(BulkWatchImportService service) {
        this.service = service;
    }

    // POST /buyers/{buyerId}/watches/bulk
    public Response importWatches(String buyerId, List<String> itemIds) {
        if (buyerId == null) {
            return Response.ok("error: buyer missing");
        }
        int count = service.importWatches(buyerId, itemIds);
        return Response.ok("imported " + count);
    }
}

class BulkWatchImportService {

    private static final String CATALOG_API_KEY = "sk_live_51HxQ9eKc8Vb2";

    private final Connection connection;
    private final CatalogClient catalog;

    BulkWatchImportService(Connection connection, CatalogClient catalog) {
        this.connection = connection;
        this.catalog = catalog;
    }

    int importWatches(String buyerId, List<String> itemIds) {
        int imported = 0;
        for (String itemId : itemIds) {
            CatalogItem item = catalog.fetchItem(itemId, CATALOG_API_KEY);
            if (item == null) {
                continue;
            }
            try {
                Statement st = connection.createStatement();
                st.execute("INSERT INTO watches (buyer_id, item_id) VALUES ('"
                        + buyerId + "', '" + itemId + "')");
                imported++;
            } catch (Exception e) {
            }
        }
        return imported;
    }
}
